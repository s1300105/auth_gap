UP_CRED = re.compile(r"(TOKEN|API[_-]?KEY|APIKEY|ACCESS[_-]?KEY|CLIENT[_-]?SECRET|(^|_)PAT($|_)|CREDENTIALS?$|SECRET[_-]?KEY|_SECRET$|PASSWORD|PASSWD|BEARER)", re.I)
INBOUND = re.compile(r"(MCP|JWT|SESSION|SIGNING|ENCRYPT|CSRF|FLASK_SECRET|DJANGO_SECRET|APP_SECRET|ADMIN_PASSWORD|AUTH_TOKEN_SECRET)", re.I)
LLM = re.compile(r"(OPENAI|ANTHROPIC|CLAUDE|GEMINI|GOOGLE_API_KEY|GROQ|MISTRAL|COHERE|DEEPSEEK|OPENROUTER|TOGETHER|HF_|HUGGING|AZURE_OPENAI|PERPLEXITY|XAI|DASHSCOPE|MOONSHOT|ZHIPU|QWEN|LLM|OLLAMA|VOYAGE|JINA|FIREWORKS|REPLICATE|LANGSMITH|LANGCHAIN|LANGFUSE|OPENAI|MINIMAX|KIMI|SILICONFLOW|ARK_|VOLC|BAICHUAN|LITELLM|AI_API_KEY|MODEL)", re.I)
SETTINGS_FIELD = re.compile(r"(token|api_key|apikey|access_key|client_secret|secret_key|password|credentials?)$", re.I)


def env_creds(t, reach_only):
    d = json.load(open(os.path.join(S, "out", t + ".json")))
    files = d["files"]
    reach = None
    if reach_only:
        g = build_graph(files)
        reach = closure(g, summ[t]["roots"]) if summ[t]["roots"] else set()
    names = set()
    for f, r in files.items():
        if r["cls"] != "src" or "skill-flow/data" in f:
            continue
        if reach is not None and f not in reach:
            continue
        for name, _ in r["env"]:
            if UP_CRED.search(name) and not INBOUND.search(name):
                names.add(name)
        for name, _ in r["settings_fields"]:
            if SETTINGS_FIELD.search(name) and not INBOUND.search(name):
                names.add("settings:" + name)
    return names


def main():
    out = {}
    for t in sorted(summ):
        st = summ[t]["status"]
        cat = MANUAL.get(t, ("M", [], "") if t in M_TREES else ("N", [], ""))
        ea = env_creds(t, False)
        er = env_creds(t, True)
        out[t] = {"status": st, "mcp_ast": summ[t]["mcp_tree_ast"], "cat": cat[0], "prov": cat[1], "rw": cat[2],
                  "env_any": sorted(ea), "env_reach": sorted(er),
                  "env_any_nonllm": sorted(n for n in ea if not LLM.search(n)),
                  "env_reach_nonllm": sorted(n for n in er if not LLM.search(n))}
    json.dump(out, open(os.path.join(S, "final.json"), "w"), indent=1)

    def report(label, pred):
        print(f"\n=== denominator: {label}")
        tot = Counter()
        for s in ("v2", "v3", "v4"):
            ts = [t for t in out if t.startswith(s + "-") and pred(out[t])]
            c = Counter(out[t]["cat"] for t in ts)
            a_w = sum(1 for t in ts if out[t]["cat"] == "A" and out[t]["rw"].startswith("W"))
            a_r = sum(1 for t in ts if out[t]["cat"] == "A" and out[t]["rw"] == "R")
            a_u = sum(1 for t in ts if out[t]["cat"] == "A" and out[t]["rw"].startswith("U"))
            no_up = [t for t in ts if out[t]["cat"] not in ("A", "Aid", "B")]
            e_any = sum(1 for t in no_up if out[t]["env_any"])
            e_reach = sum(1 for t in no_up if out[t]["env_reach"])
            e_reach_nonllm = sum(1 for t in no_up if out[t]["env_reach_nonllm"])
            e_any_nonllm = sum(1 for t in no_up if out[t]["env_any_nonllm"])
            row = dict(n=len(ts), A=c["A"], A_write=a_w, A_read=a_r, A_undet=a_u, Aid=c["Aid"], B=c["B"], C=c["C"], M=c["M"], N=c["N"],
                       E_any=e_any, E_reach=e_reach, E_any_nonllm=e_any_nonllm, E_reach_nonllm=e_reach_nonllm)
            tot.update(row)
            print(s, row)
        print("all", dict(tot))

    report("post_fetch mcp_server (project definition)", lambda r: r["status"] == "mcp_server")
    report("AST tool registration (this script)", lambda r: r["mcp_ast"])
    report("all local trees", lambda r: True)
    # providers among A (post-fetch MCP)
    pc = Counter()
    for t, r in out.items():
        if r["status"] == "mcp_server" and r["cat"] == "A":
            for p in r["prov"]:
                pc[p.split("(")[0]] += 1
    print("\nproviders among A (mcp_server):", dict(pc.most_common()))
    pc2 = Counter()
    for t, r in out.items():
        if r["status"] == "mcp_server" and r["cat"] in ("A", "Aid"):
            for p in r["prov"]:
                pc2[p.split("(")[0]] += 1
    print("providers among A+Aid (mcp_server):", dict(pc2.most_common()))


if __name__ == "__main__":
    main()
